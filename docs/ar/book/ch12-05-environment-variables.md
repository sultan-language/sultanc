# متغيرات البيئة

في النصوص اللاتينية حروف كبيرة وصغيرة، و `Sultan` و `sultan` كلمتان مختلفتان عند `ابحث`. وبعض المستخدمين يريدون أن يتجاهل
البحث هذا الفرق. والخيار من هذا النوع، الذي يضبطه المستخدم مرة ويبقيه، يُعطى كثيرا *متغير بيئة* (environment variable):
إعدادا مسمّى يمرره الطرفية إلى كل برنامج تشغّله.

## دالة بحث ثانية

المكتبة أولا، والاختبار أولا. تحوّل `حول_النص_إلى_صغير_أسكي` من `المكتبة.نصوص.تحويل` الحروف اللاتينية الكبيرة في النص إلى
صغيرة، فمقارنة الأشكال الصغيرة تتجاهل الفرق. وتأخذ الإعدادات حقلا ثالثا، `تجاهل_الحالة`، تتلقاه `ابن_الإعدادات` من مستدعيها.
وهذه المكتبة كاملة:

<sub>القائمة 12-8: البحث دون مراعاة حالة الحروف. الملف: src/search.sn</sub>

```sultan test
استورد المكتبة.تصحيح.تأكد;
استورد المكتبة.نصوص.بحث;
استورد المكتبة.نصوص.تحويل;

عام نوع إعدادات = { المطلوب: نص, مسار_الملف: نص, تجاهل_الحالة: منطقي };

عام عرف ابن_الإعدادات(الوسائط: &متجه نص, تجاهل_الحالة: منطقي): نتيجة إعدادات, نص {
    إذا (الطول(*الوسائط) < 3) {
        أرجع نتيجة.خطأ("الاستخدام: باحث المطلوب الملف");
    };
    دع الإعدادات: إعدادات = {
        المطلوب: استنسخ(&(*الوسائط)[1]),
        مسار_الملف: استنسخ(&(*الوسائط)[2]),
        تجاهل_الحالة: تجاهل_الحالة
    };
    أرجع نتيجة.نجح(الإعدادات);
}

عام عرف ابحث(المطلوب: &نص, المحتوى: &نص): متجه نص {
    دع الموجود: متغير متجه نص = [];
    دع الكل: متجه نص = أسطر_النص(المحتوى);
    لكل (دع ع: متغير عدد = 0; ع < الطول(الكل); ع := ع + 1) {
        إذا (يحتوي_نصا(&الكل[ع], المطلوب)) {
            ألحق(الموجود, استنسخ(&الكل[ع]));
        };
    }
    أرجع الموجود;
}

عام عرف ابحث_دون_مراعاة_الحالة(المطلوب: &نص, المحتوى: &نص): متجه نص {
    دع المطلوب_صغيرا: نص = حول_النص_إلى_صغير_أسكي(المطلوب);
    دع الموجود: متغير متجه نص = [];
    دع الكل: متجه نص = أسطر_النص(المحتوى);
    لكل (دع ع: متغير عدد = 0; ع < الطول(الكل); ع := ع + 1) {
        إذا (يحتوي_نصا(&حول_النص_إلى_صغير_أسكي(&الكل[ع]), &المطلوب_صغيرا)) {
            ألحق(الموجود, استنسخ(&الكل[ع]));
        };
    }
    أرجع الموجود;
}

@@اختبار
عرف نتيجة_واحدة(): فراغ {
    دع المحتوى: نص = "بطيء وثابت.\nآمن، سريع، منتج.\nاختر ثلاثة.";
    دع الموجود: متجه نص = ابحث(&"سريع", &المحتوى);
    تأكد(الطول(الموجود) == 1);
    تأكد(الموجود[0] == "آمن، سريع، منتج.");
}

@@اختبار
عرف مع_مراعاة_الحالة(): فراغ {
    دع المحتوى: نص = "Fast cars.\nsafe, fast, productive.\nFASTER still.";
    دع الموجود: متجه نص = ابحث(&"fast", &المحتوى);
    تأكد(الطول(الموجود) == 1);
}

@@اختبار
عرف دون_مراعاة_الحالة(): فراغ {
    دع المحتوى: نص = "Fast cars.\nsafe, fast, productive.\nFASTER still.";
    دع الموجود: متجه نص = ابحث_دون_مراعاة_الحالة(&"fast", &المحتوى);
    تأكد(الطول(الموجود) == 3);
}

@@اختبار
عرف وسائط_ناقصة(): فراغ {
    دع الوسائط: متجه نص = ["باحث", "fast"];
    طابق ابن_الإعدادات(&الوسائط, خطأ) مع
    | نتيجة.نجح(_) -> { تأكد(خطأ); }
    | نتيجة.خطأ(_) -> { };
}
```

```text
نجح نتيجة_واحدة
نجح مع_مراعاة_الحالة
نجح دون_مراعاة_الحالة
نجح وسائط_ناقصة

نتيجة الاختبارات: ناجحة. نجح: 4؛ فشل: 0؛ استُبعد: 0
```

يفحص `مع_مراعاة_الحالة` أن `ابحث` ما زالت تراعي الحالة، ويفحص `دون_مراعاة_الحالة` أن الدالة الجديدة لا تراعيها. ويفحص
`وسائط_ناقصة` أن `ابن_الإعدادات` ترفض متجها بلا اسم ملف.

ولا تغيّر `حول_النص_إلى_صغير_أسكي` إلا الحروف اللاتينية من A إلى Z. وليس في العربية حروف كبيرة، فيُبحث في النص العربي بالطريقة
نفسها في الدالتين.

## قراءة متغير البيئة

تقرأ `متغير_بيئة(&الاسم)` من `المكتبة.بيئة.بيئة_العمل` متغيرا. وقد لا يكون المتغير مضبوطا، وقد تفشل قراءة البيئة، فترجع
`نتيجة` تحمل `اختياري`. ولا يهم هنا إلا هل هو مضبوط:

```sultan ignore
عرف طُلب_تجاهل_الحالة(): منطقي {
    طابق متغير_بيئة(&"IGNORE_CASE") مع
    | نتيجة.نجح(اختياري.موجود(_)) -> { أرجع صواب; }
    | نتيجة.نجح(اختياري.لا_شيء) -> { أرجع خطأ; }
    | نتيجة.خطأ(_) -> { أرجع خطأ; };
}
```

والنمط `نتيجة.نجح(اختياري.موجود(_))` يطابق مستويين معا: `نجح` تحمل `موجود`.

وتمرر `الرئيسية` الجواب إلى `ابن_الإعدادات`، وتختار `شغل` دالة البحث حسب الإعداد:

```sultan ignore
    دع الموجود: متجه نص;
    إذا (الإعدادات.تجاهل_الحالة) {
        الموجود := ابحث_دون_مراعاة_الحالة(&الإعدادات.المطلوب, &المحتوى);
    } وإلا {
        الموجود := ابحث(&الإعدادات.المطلوب, &المحتوى);
    };
```

```console
$ supax run -- sultan names.txt
the sultan compiler
$ IGNORE_CASE=1 supax run -- sultan names.txt
Sultan language
the sultan compiler
SULTAN in capitals
```

حُذفت أسطر `يبني` و `انتهى` و `يشغّل` من هذه المخرجات. والملف `names.txt` فيه الأسطر الثلاثة. و `IGNORE_CASE=1` قبل الأمر
يضبط المتغير لذلك الأمر وحده، في طرفيات لينكس وماك.
