# تطوير المكتبة بالاختبارات

البحث نفسه قلب البرنامج. ويكتبه هذا القسم بعكس المعتاد: الاختبار أولا، ثم الشيفرة. ولطريقة العمل هذه، *التطوير الموجَّه
بالاختبارات* (test-driven development)، ثلاث خطوات:

1. اكتب اختبارا يفشل، لأن الشيفرة التي يختبرها غير موجودة أو لا تعمل بعد. شغّله وشاهده يفشل للسبب الذي تتوقعه.
2. اكتب ما يكفي من الشيفرة لينجح الاختبار.
3. حسّن الشيفرة، وشغّل الاختبارات لتتأكد أنه ما زال ينجح.

## اختبار يفشل

يقول الاختبار ما ينبغي أن تفعله `ابحث`: تعطيها مطلوبا ونصا، فترجع الأسطر التي فيها المطلوب. أضفه إلى `src/search.sn`، مع
`ابحث` لا ترجع شيئا بعد:

<sub>القائمة 12-5: اختبار، و `ابحث` لا تعمل بعد. الملف: src/search.sn</sub>

```sultan ignore
استورد المكتبة.تصحيح.تأكد;
استورد المكتبة.نصوص.مقارنة;

عام عرف ابحث(المطلوب: &نص, المحتوى: &نص): متجه نص {
    دع الموجود: متجه نص = [];
    أرجع الموجود;
}

@@اختبار
عرف نتيجة_واحدة(): فراغ {
    دع المحتوى: نص = "بطيء وثابت.\nآمن، سريع، منتج.\nاختر ثلاثة.";
    دع الموجود: متجه نص = ابحث(&"سريع", &المحتوى);
    تأكد(الطول(الموجود) == 1);
    تأكد(نصان_متساويان(&الموجود[0], &"آمن، سريع، منتج."));
}
```

## إنجاح الاختبار

تمر `ابحث` على أسطر `المحتوى`، وتُبقي ما فيه `المطلوب`. وتقسم `أسطر_النص` من `المكتبة.نصوص.تحويل` النص إلى أسطره، وتقول
`يحتوي_نصا` من `المكتبة.نصوص.بحث` هل يحتوي نص نصا آخر:

<sub>القائمة 12-6: `ابحث` تعمل. الملف: src/search.sn</sub>

```sultan test
استورد المكتبة.تصحيح.تأكد;
استورد المكتبة.نصوص.مقارنة;
استورد المكتبة.نصوص.بحث;
استورد المكتبة.نصوص.تحويل;

عام نوع إعدادات = { المطلوب: نص, مسار_الملف: نص };

عام عرف ابن_الإعدادات(الوسائط: &متجه نص): نتيجة إعدادات, نص {
    إذا (الطول(*الوسائط) < 3) {
        أرجع نتيجة.خطأ("الاستخدام: باحث المطلوب الملف");
    };
    دع الإعدادات: إعدادات = { المطلوب: استنسخ(&(*الوسائط)[1]), مسار_الملف: استنسخ(&(*الوسائط)[2]) };
    أرجع نتيجة.نجح(الإعدادات);
}

عام عرف ابحث(المطلوب: &نص, المحتوى: &نص): متجه نص {
    دع الموجود: متغير متجه نص = [];
    دع الكل: متجه نص = أسطر_النص(المحتوى);
    لكل (دع ع: متغير عدد = 0; ع < الطول(الكل); ع := ع + 1) {
        إذا (يحتوي_نصا(&الكل[ع], المطلوب)) {
            ألحق(الموجود, استنسخ(&الكل[ع]));
        };
    }
    أرجع الموجود;
}

@@اختبار
عرف نتيجة_واحدة(): فراغ {
    دع المحتوى: نص = "بطيء وثابت.\nآمن، سريع، منتج.\nاختر ثلاثة.";
    دع الموجود: متجه نص = ابحث(&"سريع", &المحتوى);
    تأكد(الطول(الموجود) == 1);
    تأكد(نصان_متساويان(&الموجود[0], &"آمن، سريع، منتج."));
}
```

```text
نجح نتيجة_واحدة

نتيجة الاختبارات: ناجحة. نجح: 1؛ فشل: 0؛ استُبعد: 0
```

والأسطر في `الموجود` نسخ مصنوعة بـ `استنسخ`: فـ `الكل` متجه محلي يُحرَّر حين ترجع `ابحث`، والمتجه المرجَع يجب أن يملك نصوصه.

## استخدام `ابحث` في `شغل`

تطبع `شغل` الآن الأسطر التي وجدتها `ابحث` بدلا من النص كله:

<sub>القائمة 12-7: `شغل` مع البحث. الملف: src/main.sn</sub>

```sultan ignore
عرف شغل(الإعدادات: &إعدادات): نتيجة عدد, نص {
    دع المحتوى: نص;
    طابق اقرأ_الملف(&(*الإعدادات).مسار_الملف) مع
    | نتيجة.نجح(النص) -> { المحتوى := النص; }
    | نتيجة.خطأ(_) -> { أرجع نتيجة.خطأ(صل_نصين(&"تعذرت قراءة ", &(*الإعدادات).مسار_الملف)); };
    دع الموجود: متجه نص = ابحث(&(*الإعدادات).المطلوب, &المحتوى);
    لكل (دع ع: متغير عدد = 0; ع < الطول(الموجود); ع := ع + 1) {
        اطبع_سطر(استنسخ(&الموجود[ع]));
    }
    أرجع نتيجة.نجح(0);
}
```

```console
$ supax run -- الحجر poem.txt

       يشغّل build/bin/باحث
النهر لا يسأل الحجر
اسأل الحجر، فلن يقول شيئا.
```

يعمل البرنامج. ويضيف القسم التالي خيارا يغيّر طريقة البحث.
