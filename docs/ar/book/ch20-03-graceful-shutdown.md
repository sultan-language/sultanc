# الإيقاف الآمن

الضغط على Ctrl-C يوقف الخادم فورا، ولو في منتصف رد. والخادم الذي يتوقف *بنظام* (gracefully) يكفّ عن قبول اتصالات جديدة، ويترك
العمال ينهون ما بأيديهم، ثم ينتهي.

وفي المجموعة كل ما يلزم لذلك. فـ `استقبل` لدى العامل تفشل حين لا يبقى لقناته مرسلون، فتنتهي حلقته. فلإيقاف العمال حرّر
المرسلين، ولانتظارهم انضم إلى خيوطهم.

ولإبقاء المثال قصيرا يتوقف هذا الخادم بعد خدمة اتصالين. وتنتقل حلقة القبول إلى `اخدم`، التي تأخذ متجه المرسلين بالقيمة. فحين
ترجع `اخدم` يخرج المتجه من نطاقه، ومعه كل مرسل:

<sub>القائمة 20-5: التوقف بعد اتصالين. الملف: src/main.sn</sub>

```sultan ignore
عرف اخدم(المستمع: &مستمع_تي_سي_بي, المرسلون: متجه مرسل<تدفق_تي_سي_بي>, الحد: عدد): فراغ {
    ليكن التالي: متغير عدد = 0;
    لكل (ليكن المخدوم: متغير عدد = 0; المخدوم < الحد; المخدوم := المخدوم + 1) {
        طابق اقبل_تي_سي_بي(المستمع) مع
        | نتيجة.نجح(التدفق) -> {
            طابق أرسل(&المرسلون[التالي], التدفق) مع
            | نتيجة.نجح(_) -> { }
            | نتيجة.خطأ(_) -> { };
            التالي := (التالي + 1) % الطول(المرسلون);
        }
        | نتيجة.خطأ(_) -> { };
    }
    اطبع_سطر("يتوقف الخادم.");
}

عرف الرئيسية(): عدد {
    ليكن العنوان: عنوان_مقبس;
    طابق حلل_عنوان_مقبس(&"127.0.0.1:7878") مع
    | نتيجة.نجح(ع_م) -> { العنوان := ع_م; }
    | نتيجة.خطأ(_) -> { أرجع 1; };
    ليكن المستمع: مستمع_تي_سي_بي;
    طابق اربط_مستمع_تي_سي_بي(&العنوان, 128) مع
    | نتيجة.نجح(م) -> { المستمع := م; }
    | نتيجة.خطأ(الرمز) -> { أرجع الرمز; };

    ليكن العمال: متغير متجه مقبض_خيط = [];
    ليكن المرسلون: متغير متجه مرسل<تدفق_تي_سي_بي> = [];
    لكل (ليكن الرقم: متغير عدد = 0; الرقم < 4; الرقم := الرقم + 1) {
        طابق ابدأ_عاملا(الرقم, &متغير العمال) مع
        | اختياري.موجود(الصادر) -> { ألحق(المرسلون, الصادر); }
        | اختياري.لا_شيء -> { أرجع 1; };
    }

    اخدم(&المستمع, المرسلون, 2);
    لكل (ليكن ع: متغير عدد = 0; ع < الطول(العمال); ع := ع + 1) {
        انضم(&متغير العمال[ع]);
    }
    أرجع 0;
}
```

وبعد أن ترجع `اخدم` تنضم `الرئيسية` إلى كل عامل. فينهي كل منهم الاتصال الذي قد يعالجه، ويجد قناته فارغة بلا مرسلين،
فيتوقف:

```console
$ curl -s -o /dev/null -w '%{http_code}\n' http://127.0.0.1:7878/
200
$ curl -s -o /dev/null -w '%{http_code}\n' http://127.0.0.1:7878/other
404
```

```text
العامل 0 استلم اتصالا.
GET / HTTP/1.1
يتوقف الخادم.
العامل 1 استلم اتصالا.
GET /other HTTP/1.1
العامل 1 توقف.
العامل 0 توقف.
العامل 2 توقف.
العامل 3 توقف.
```

تعمل الخيوط في الوقت نفسه، فقد يتغير ترتيب هذه الأسطر من تشغيل لآخر. وما لا يتغير: يُجاب كل اتصال، ويتوقف كل عامل، وينتهي
البرنامج بالحالة 0.

ولم يُكتب شيء في هذا الإيقاف خصيصا له. فالعمال يتوقفون لأن الملكية تحرر المرسلين، والقناة تبلّغ بذلك. وهي القاعدة نفسها التي
تحرر نصا في نهاية كتلته.

## إلى أين بعد ذلك

هذه نهاية الكتاب. رأيت اللغة كلها: القيم والأنواع، والملكية، والسجلات والتعدادات، والوحدات والحزم، والأخطاء، والأنواع العامة
والـ closures، والاختبارات، والخيوط، والمكتبة القياسية، و C، والمنصات. وبعض طرق المتابعة:

- وسّع الخادم: اخدم أي ملف من مجلد، واضبط ترويسة `Content-Type` حسب امتداد الملف، وحدّ حجم الطلب، وأضف مسارا يجيب بـ JSON.
- اقرأ [المرجع](../index.md#المرجع)، الذي يحدد اللغة والمكتبة القياسية بدقة.
- تصفّح مجلد `examples` في المستودع: نحو 500 برنامج صغير، كل منها بالعربية والإنجليزية.
- وتسرد ملاحق هذا الكتاب الكلمات المفتاحية باللغتين، والعوامل، ورسائل التشخيص، ووحدات المكتبة القياسية.
