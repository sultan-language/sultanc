# الخرائط والمجموعات

يجد المتجه القيمة بموضعها. وكثيرا ما تريد أن تجدها بشيء آخر: عدد مرات كلمة بالكلمة، أو مستخدما باسمه. فـ*الخريطة* (map) تخزّن
أزواجا من *مفتاح* و*قيمة*، وتجد القيمة من المفتاح. و*المجموعة* (set) تخزّن مفاتيح وحدها، كلا منها مرة واحدة.

وفي `المكتبة.مجموعات` عدة منها:

| المجموعة | الوحدة | فيمَ تُستخدم |
| --- | --- | --- |
| `خريطة_نص_فهرس` | `المكتبة.مجموعات.خريطة_نص_فهرس` | مفاتيح نصية إلى قيم `طبيعي`: للعدّ والترقيم |
| `مجموعة_نص` | `المكتبة.مجموعات.خريطة_نص_فهرس` | نصوص مميزة |
| `خريطة_شجرة_بي<م, ق>` | `المكتبة.مجموعات.خريطة_شجرية` | أي مفاتيح، محفوظة مرتبة |
| `الخريطة<م, ق>` | `المكتبة.مجموعات.خريطة` | أي مفاتيح، بالتجزئة: أسرع بحث |
| `مجموعة<ن>` | `المكتبة.مجموعات.مجموعة` | قيم مميزة من أي نوع، بالتجزئة |
| `طابور_ذو_نهايتين<ن>` | `المكتبة.مجموعات.رتل_مزدوج` | طابور يُضاف إليه ويؤخذ منه من الطرفين |

ودوال كل منها مسماة باسمها: `ابحث_في_خريطة_نص_فهرس` و `أضف_إلى_مجموعة` و `خذ_من_أمام_طابور_ذو_نهايتين`.

ويتكرر نمطان فيها كلها. البحث عن مفتاح يرجع `اختياري`، لأن المفتاح قد لا يوجد. والإضافة ترجع `نتيجة`، لأنها قد تحتاج إلى
ذاكرة أكثر، وقد تنفد الذاكرة. وتنهي الأمثلة هنا البرنامج بالحالة 1 في هذه الحالة.

## عدّ الكلمات

تربط `خريطة_نص_فهرس` نصا بقيمة `طبيعي`، فهي أداة العدّ:

```sultan
استورد المكتبة.دخ.اطبع_سطر;
استورد المكتبة.نصوص.أعداد;
استورد المكتبة.نصوص.تحويل;
استورد المكتبة.مجموعات.خريطة_نص_فهرس;

عرف الرئيسية(): عدد {
    ليكن النص: نص = "في البيت قط في الحديقة كلب في الشارع طائر";
    ليكن الكلمات: متجه نص = قسم_النص(&النص, &" ");
    ليكن العدادات: متغير خريطة_نص_فهرس = أنشئ_خريطة_نص_فهرس();
    لكل (ليكن ع: متغير عدد = 0; ع < الطول(الكلمات); ع := ع + 1) {
        ليكن حتى_الآن: متغير طبيعي = 0;
        طابق ابحث_في_خريطة_نص_فهرس(&العدادات, &الكلمات[ع]) مع
        | اختياري.موجود(ن) -> { حتى_الآن := ن; }
        | اختياري.لا_شيء -> { };
        طابق أدرج_في_خريطة_نص_فهرس(&متغير العدادات, استنسخ(&الكلمات[ع]), حتى_الآن + 1) مع
        | نتيجة.نجح(_) -> { }
        | نتيجة.خطأ(_) -> { أرجع 1; };
    }
    طابق ابحث_في_خريطة_نص_فهرس(&العدادات, &"في") مع
    | اختياري.موجود(ن) -> { اطبع_سطر(طبيعي_إلى_نص(ن)); }
    | اختياري.لا_شيء -> { اطبع_سطر("0"); };
    اطبع_سطر(طبيعي_إلى_نص(طول_خريطة_نص_فهرس(&العدادات)));
    أرجع 0;
}
```

```text
3
7
```

لكل كلمة يبحث البرنامج عن عددها حتى الآن (و `لا_شيء` يعني صفرا)، ويخزّن العدد زائدا واحدا. وإدراج مفتاح موجود يستبدل قيمته.
ويُمرَّر المفتاح بقيمته، فتملك الخريطة نسختها الخاصة من الكلمة: `استنسخ(&الكلمات[ع])`.

## قيم مميزة، بترتيب

تحفظ `مجموعة_نص` كل نص مرة واحدة. ومع الخريطة و `رتب_نصوص` تعطي عدد الكلمات مرتبا معجميا:

```sultan
استورد المكتبة.دخ.اطبع_سطر;
استورد المكتبة.نصوص.أعداد;
استورد المكتبة.نصوص.تحويل;
استورد المكتبة.مجموعات.خريطة_نص_فهرس;
استورد المكتبة.مجموعات.ترتيب;

عرف الرئيسية(): عدد {
    ليكن النص: نص = "في البيت قط في الحديقة كلب في الشارع طائر في البيت قط";
    ليكن الكلمات: متجه نص = قسم_النص(&النص, &" ");
    ليكن العدادات: متغير خريطة_نص_فهرس = أنشئ_خريطة_نص_فهرس();
    ليكن الفريدة: متغير مجموعة_نص = أنشئ_مجموعة_نص();
    لكل (ليكن ع: متغير عدد = 0; ع < الطول(الكلمات); ع := ع + 1) {
        ليكن حتى_الآن: متغير طبيعي = 0;
        طابق ابحث_في_خريطة_نص_فهرس(&العدادات, &الكلمات[ع]) مع
        | اختياري.موجود(ن) -> { حتى_الآن := ن; }
        | اختياري.لا_شيء -> { };
        طابق أدرج_في_خريطة_نص_فهرس(&متغير العدادات, استنسخ(&الكلمات[ع]), حتى_الآن + 1) مع
        | نتيجة.نجح(_) -> { }
        | نتيجة.خطأ(_) -> { أرجع 1; };
        طابق أضف_إلى_مجموعة_نص(&متغير الفريدة, استنسخ(&الكلمات[ع])) مع
        | نتيجة.نجح(_) -> { }
        | نتيجة.خطأ(_) -> { أرجع 1; };
    }
    ليكن المرتبة: متغير متجه نص = قيم_مجموعة_نص(&الفريدة);
    رتب_نصوص(&متغير المرتبة);
    لكل (ليكن ع: متغير عدد = 0; ع < الطول(المرتبة); ع := ع + 1) {
        طابق ابحث_في_خريطة_نص_فهرس(&العدادات, &المرتبة[ع]) مع
        | اختياري.موجود(ن) -> { اطبع_سطر(صل_ثلاثة_نصوص(&المرتبة[ع], &": ", &طبيعي_إلى_نص(ن))); }
        | اختياري.لا_شيء -> { };
    }
    أرجع 0;
}
```

```text
البيت: 2
الحديقة: 1
الشارع: 1
طائر: 1
في: 4
قط: 2
كلب: 1
```

## خرائط بأي مفتاح، مرتبة

تأخذ `خريطة_شجرة_بي<م, ق>` مفاتيح من أي نوع `م` وقيما من أي نوع `ق`، وتحفظ المفاتيح مرتبة. ولتحفظها مرتبة يجب أن تقارنها،
فتعطيها المقارنة حين تنشئها: دالة ترجع عددا سالبا أو صفرا أو موجبا، كما تفعل `قارن_نصين_معجميا`. ويشرح
[الفصل العاشر](ch10-02-functions-as-values.md) تمرير الدوال قيما.

```sultan
استورد المكتبة.دخ.اطبع_سطر;
استورد المكتبة.نصوص.أعداد;
استورد المكتبة.نصوص.تحويل;
استورد المكتبة.مجموعات.خريطة_شجرية;

عرف قارن_الأعداد(الأيسر: &عدد, الأيمن: &عدد): عدد {
    أرجع *الأيسر - *الأيمن;
}

عرف الرئيسية(): عدد {
    ليكن المراتب: متغير خريطة_شجرة_بي<عدد, نص> = أنشئ_خريطة_شجرة_بي(قارن_الأعداد);
    أدرج_في_خريطة_شجرة_بي(&متغير المراتب, 30, "الثالث");
    أدرج_في_خريطة_شجرة_بي(&متغير المراتب, 10, "الأول");
    أدرج_في_خريطة_شجرة_بي(&متغير المراتب, 20, "الثاني");
    ليكن المفاتيح: متجه عدد = مفاتيح_خريطة_شجرة_بي(&المراتب);
    ليكن المعروضة: متغير متجه نص = [];
    لكل (ليكن ع: متغير عدد = 0; ع < الطول(المفاتيح); ع := ع + 1) {
        ألحق(المعروضة, عدد_إلى_نص(المفاتيح[ع]));
    }
    اطبع_سطر(صل_النصوص(&المعروضة, &" "));
    طابق ابحث_في_خريطة_شجرة_بي(&المراتب, &20) مع
    | اختياري.موجود(الاسم) -> { اطبع_سطر(الاسم); }
    | اختياري.لا_شيء -> { اطبع_سطر("غير موجود"); };
    أرجع 0;
}
```

```text
10 20 30
الثاني
```

أُدرجت المفاتيح 30 ثم 10 ثم 20، وخرجت مرتبة. وتنسخ `مفاتيح_خريطة_شجرة_بي` المفاتيح في متجه جديد، فتحتاج إلى نوع مفتاح بسيط
مثل `عدد`؛ ويرفضها المترجم عند الاستدعاء مع نوع مفتاح يملك ذاكرة، ويسمّي الدالة. ولمثل هذه المفاتيح، كالنصوص، ترجع
`مفاتيح_خريطة_شجرة_بي_منسوخة` نسخة من كل مفتاح، بالترتيب.

## مجموعات من أي نوع

تحفظ `مجموعة<ن>` بالتجزئة قيما مميزة من أي نوع. ومجموعة التجزئة تجد القيمة بتحويلها إلى عدد هو *بصمتها* (hash)، فتحتاج إلى
دالتين: واحدة تحسب البصمة، وواحدة تقول هل قيمتان متساويتان:

```sultan
استورد المكتبة.دخ.اطبع_سطر;
استورد المكتبة.نصوص.أعداد;
استورد المكتبة.مجموعات.مجموعة;

عرف جزئ_العدد(القيمة: &عدد): طبيعي {
    أرجع ((*القيمة) مثل طبيعي) * 2654435761;
}

عرف عددان_متساويان(أ: &عدد, ب: &عدد): منطقي {
    أرجع *أ == *ب;
}

عرف الرئيسية(): عدد {
    ليكن المرئية: متغير مجموعة<عدد> = أنشئ_مجموعة(جزئ_العدد, عددان_متساويان);
    ليكن القيم: متجه عدد = [4, 8, 4, 15, 8, 16, 4];
    لكل (ليكن ع: متغير عدد = 0; ع < الطول(القيم); ع := ع + 1) {
        طابق أضف_إلى_مجموعة(&متغير المرئية, القيم[ع]) مع
        | نتيجة.نجح(_) -> { }
        | نتيجة.خطأ(_) -> { أرجع 1; };
    }
    اطبع_سطر(طبيعي_إلى_نص(طول_مجموعة(&المرئية)));
    إذا (تحتوي_مجموعة(&المرئية, &15)) {
        اطبع_سطر("فيها 15");
    };
    أرجع 0;
}
```

```text
4
فيها 15
```

وتعمل `الخريطة<م, ق>` في `المكتبة.مجموعات.خريطة` بالطريقة نفسها، مع دالة بصمة ودالة مساواة لمفاتيحها.

## الطوابير

`طابور_ذو_نهايتين` يضيف القيم ويحذفها من الطرفين. وأخذ ما أُضيف في الخلف من الأمام يصنع طابورا: الداخل أولا يخرج أولا.

```sultan
استورد المكتبة.دخ.اطبع_سطر;
استورد المكتبة.نصوص.أعداد;
استورد المكتبة.مجموعات.رتل_مزدوج;

عرف الرئيسية(): عدد {
    ليكن المنتظرون: متغير طابور_ذو_نهايتين<عدد> = أنشئ_طابور_ذو_نهايتين();
    أضف_خلف_طابور_ذو_نهايتين(&متغير المنتظرون, 101);
    أضف_خلف_طابور_ذو_نهايتين(&متغير المنتظرون, 102);
    أضف_خلف_طابور_ذو_نهايتين(&متغير المنتظرون, 103);
    طالما (طول_طابور_ذو_نهايتين(&المنتظرون) > 0) {
        طابق خذ_من_أمام_طابور_ذو_نهايتين(&متغير المنتظرون) مع
        | اختياري.موجود(التذكرة) -> { اطبع_سطر(عدد_إلى_نص(التذكرة)); }
        | اختياري.لا_شيء -> { };
    }
    أرجع 0;
}
```

```text
101
102
103
```

## اختيار المجموعة

- قيم بترتيب، تُوجد بموضعها: `متجه`.
- نص إلى عدد: `خريطة_نص_فهرس`. ونصوص مميزة: `مجموعة_نص`.
- مفاتيح محفوظة مرتبة: `خريطة_شجرة_بي`. وأسرع بحث بالمفتاح: `الخريطة`.
- قيم مميزة من أي نوع: `مجموعة`. وعمل ينتظر دوره: `طابور_ذو_نهايتين`.

## الخلاصة

- يحمل `متجه ن` القيم بترتيب: `ألحق` تضيف، و `م[ع]` تقرأ، و `الطول` تعدّ.
- `نص` كتابة UTF-8: `الطول` تعدّ البايتات، و `عدد_محارف_النص` المحارف، والمواضع بايتات.
- تُوصل النصوص بـ `صل_نصين` و `صل_ثلاثة_نصوص` و `صل_النصوص`، وتقارَن بـ `==`، وتُقطَّع بـ `قص_النص` و `قسم_النص`
  و `استبدل_في_النص` و `مقطع_نص`.
- الخرائط تجد القيم بالمفتاح، والمجموعات تحفظ قيما مميزة. والبحث يرجع `اختياري`، والإدراج `نتيجة`.
- المجموعات من أي نوع تأخذ دوالا لمقارنة مفاتيحها أو حساب بصمتها.

جرّب هذه التمارين:

- من متجه أعداد صحيحة، اطبع المتوسط، والوسيط (القيمة الوسطى بعد الترتيب)، والقيمة الأكثر تكرارا.
- اقرأ أسطرا بالشكل `القسم الاسم` واطبع أسماء كل قسم مرتبة.
- اقرأ جملة واطبع كل كلمة فيها مع عدد محارفها.
